# A property that can fail (Apply, Analyse)

Your unit test checks a lab of three steps and a lab of one. A property checks every lab of one to six steps.
Create `tests/fuzz/step-position.fuzz.test.ts`:

~~~ts
/**
 * A property of step positions (R-013): for every lab of one to six steps, not just three.
 * Replay a failure: FUZZ_SEED=<seed> FUZZ_PATH=<path> pnpm test:fuzz
 */
import fc from "fast-check";
import { expect, test } from "vitest";
import type { Course } from "@ladder/model";
import { buildSite } from "@ladder/lite";
import type { SitePage } from "@ladder/lite";
import { fuzzParameters } from "../support/fuzz.ts";

const labOf = (n: number): Course => ({
  type: "course", id: "c", title: "C", summary: "", contentMd: "# C", route: "/",
  los: [{ type: "lab", id: "l", title: "L", summary: "", contentMd: "", route: "", los: Array.from({ length: n }, (_, i) => ({ type: "step" as const, id: `S${i}`, shortTitle: `S${i}`, title: `S${i}`, contentMd: "", route: "" })) }],
});

/** The property, over any site builder, so the negative fixture can run it on a broken one. */
const labelsEveryStep = (build: (course: Course) => SitePage[]) =>
  fc.property(fc.integer({ min: 1, max: 6 }), (n) => {
    const steps = build(labOf(n)).filter((p) => p.path.startsWith("l/"));
    expect(steps).toHaveLength(n);
    steps.forEach((step, i) => {
      if (n > 1) expect(step.html).toContain(`<p class="position">Step ${i + 1} of ${n}</p>`);
      else expect(step.html).not.toContain('class="position"');
    });
  });

test("R-013: page i of a lab of n steps says Step i of n when n > 1, and nothing when n = 1", () => {
  fc.assert(labelsEveryStep(buildSite), fuzzParameters());
});

test("the step-position property can fail: it catches a site that labels a lab of one step", () => {
  const eager = (course: Course) => buildSite(course).map((p) => (p.path === "l/index.html" && !p.html.includes("position") ? { ...p, html: p.html.replace("<ol>", '<p class="position">Step 1 of 1</p><ol>') } : p));
  expect(fc.check(labelsEveryStep(eager), fuzzParameters()).failed).toBe(true);
});
~~~
