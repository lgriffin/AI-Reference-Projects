## Bind the steps

In `tests/bdd/steps/site/static-site.steps.ts`, add a `Rule` just before the final `});` of
`describeFeature`. The Given and When are the same as R-010's, so reuse them:

~~~ts
  Rule("When a lab has more than one step, the site shall label each step page with its position, such as Step 2 of 3.", ({ RuleScenario }) => {
    RuleScenario("Each step knows where it is", ({ Given, When, Then }) => {
      Given("a course with a lab whose steps are {string}", (_ctx, names: string) => {
        const steps: Lab["los"] = list(names).map((name) => ({ type: "step", id: name, shortTitle: name, title: name, contentMd: `# ${name}`, route: "" }));
        course = courseOf({ ...lo("book-1"), type: "lab", los: steps });
      });
      When("the site is built", built);
      Then("the step page {string} says {string}", (_ctx, path: string, text: string) => {
        expect(page(`book-1/${path}`)).toContain(text);
      });
    });
  });
~~~
