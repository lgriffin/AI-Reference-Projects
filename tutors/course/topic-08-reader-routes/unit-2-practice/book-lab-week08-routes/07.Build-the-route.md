test("a wall holds every learning object of one kind, and only talks, notes and labs have one", () => {

  assert.deepEqual(wall(tree, "note").map((lo) => lo.title), ["Reading Guide"]);

~~~

Then the code, at the end of `apps/reader/src/lib/course/lo-tree.ts`:

~~~ts
/** The kinds that have a wall: every talk, note or lab of a course on one page. Upstream: createWalls. */
export const WALL_KINDS = ["talk", "note", "lab"] as const;
export type WallKind = (typeof WALL_KINDS)[number];
export const isWallKind = (kind: string): kind is WallKind => (WALL_KINDS as readonly string[]).includes(kind);

/** Every learning object of one kind that the course holds, in the order the course reads (R-022). */
export function wall(tree: CourseTree, kind: WallKind): PageLo[] {
  return [...tree.loIndex.values()].filter((lo) => lo.type === kind);
}
~~~

## 2. The request: a page, or a 404 message

A route decides nothing, so the course service must answer for a kind that has no wall. Test first, at the
end of `tests/unit/reader/course-service.test.ts` (add `requestWall` to its import):

~~~ts
// Added in Lab 8 with R-022.
test("requestWall answers with the wall, or says whether the course or the kind is missing", async () => {
  const { fetch } = fakeFetch();
  const labs = await requestWall("lab", "sample", fetch);
  assert.ok("found" in labs && labs.found.kind === "lab" && labs.tree.courseId === "sample");
  assert.deepEqual("found" in labs && labs.found.los.map((lo) => lo.title), ["Lab-01-First-Steps"]);
  assert.deepEqual(await requestWall("quiz", "sample", fetch), { notFound: 'Not found: a wall of "quiz"' });
  assert.deepEqual(await requestWall("lab", "missing", fetch), { notFound: 'Not found: the course "missing"' });
});
~~~

In `apps/reader/src/lib/course/course-service.ts`, extend the import from `./lo-tree.ts` with `isWallKind`,
`wall` and `type WallKind`, and add at the end:

~~~ts
/** A wall: one kind, and every learning object of that kind in the course. */
export type Wall = { kind: WallKind; los: PageLo[] };

/** The wall of this kind for this course, for the wall page (R-022). The URL's kind is a string until checked. */
export async function requestWall(kind: string, courseId: string, fetch: Fetch): Promise<PageRequest<Wall>> {
  const tree = await readCourse(courseId, fetch);
  if (!tree) return missingCourse(courseId);
  return isWallKind(kind) ? { tree, found: { kind, los: wall(tree, kind) } } : { notFound: `Not found: a wall of "${kind}"` };
}
~~~

## 3. The route

Make the folder (quote it: the shell reads brackets):

~~~bash
mkdir -p 'apps/reader/src/routes/(course-reader)/wall/[type]/[courseid]'
~~~

`+page.ts` in it, modelled on the topic route's load:

~~~ts
import { requestWall } from "#lib/course/course-service.ts";
import { currentCourse, currentLo } from "#lib/runes/index.svelte.ts";
import { found } from "../../../found.ts";
import type { PageLoad } from "./$types";

// /wall/<type>/<courseid>: every talk, note or lab of the course (R-022). Upstream has the same route.
export const load: PageLoad = async ({ params, fetch }) => {
  const { tree, found: wall } = found(await requestWall(params.type, params.courseid, fetch));
  currentCourse.value = tree;
  currentLo.value = tree.course;
  return { course: tree.course, wall };
};
~~~

`+page.svelte` beside it hands the data to one component:

~~~svelte
<script lang="ts">
  import Wall from "#lib/ui-components/layout/Wall.svelte";
  import type { PageProps } from "./$types";

  let { data }: PageProps = $props();
</script>

<Wall course={data.course} wall={data.wall} />
~~~

and the component, `apps/reader/src/lib/ui-components/layout/Wall.svelte`, reuses the card grid:

~~~svelte
<!--
  A wall: every learning object of one kind in the course, as cards (R-022). Mirrors upstream Wall
  (packages/svelte/ui-components/src/learning-objects/layout/Wall.svelte).
-->

  import type { Course } from "@ladder/model";

~~~

## 4. A way in

A student cannot use a page they must type the address of. Until a link exists, the wall is inventory:
finished work delivering nothing. In `apps/reader/src/lib/ui-navigators/TutorsShell.svelte`, import the
kinds:

~~~ts
  import { WALL_KINDS } from "../course/lo-tree.ts";
~~~

replace the `{#if course}...{/if}` line in the header with:

~~~svelte
  {#if course}
    <a class="course" href={course.route}>{course.title}</a>
    <nav aria-label="Walls">
      {#each WALL_KINDS as kind (kind)}<a href={`/wall/${kind}/${currentCourse.value?.courseId}`}>{kind}s</a>{/each}
    </nav>
  {/if}
~~~

and add one line to its `<style>`:

~~~css
  nav { display: flex; gap: 0.75rem; margin-left: auto; }
~~~

Reload. The top bar now has **talks**, **notes** and **labs** on every course page.
