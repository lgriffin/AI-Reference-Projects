# The grading test

Save this as `variants/summit-composed/tests/compose.grading.test.ts`. It runs with `node --test` on Node 22.18
or later and needs no dependencies. It is part of your gate, so `pnpm test` must run it.

~~~ts
import { test } from "node:test";
import assert from "node:assert/strict";
import { existsSync, readFileSync, readdirSync, statSync } from "node:fs";
import { join, relative } from "node:path";

const here = join(import.meta.dirname, "..");
const variants = join(here, "..");
const rungs = ["rung-4-reader", "rung-5-proven", "rung-6-guarded", "rung-7-shipped"];

function testFiles(dir: string): string[] {
  if (!existsSync(dir)) return [];
  const out: string[] = [];
  for (const name of readdirSync(dir)) {
    const path = join(dir, name);
    if (statSync(path).isDirectory()) out.push(...testFiles(path));
    else if (/\.test\.ts$/.test(name)) out.push(path);
  }
  return out;
}

test("AC1: the composed variant has the files every variant has", () => {
  for (const file of ["README.md", "RUNG.md", "AGENTS.md", "package.json", "tests/rung-fence.test.ts"]) {
    assert.ok(existsSync(join(here, file)), `AC1: ${file} is missing. Every variant has it (see tutors/LADDER.md, "Variant conventions").`);
  }
});

test("AC3: every test of rungs 4 to 7 survives the composition", () => {
  for (const rung of rungs) {
    const root = join(variants, rung);
    assert.ok(existsSync(root), `AC3: variants/${rung} is missing. Compose from the rung folders, do not delete them.`);
    for (const file of testFiles(join(root, "tests"))) {
      const rel = relative(root, file);
      if (rel.endsWith("rung-fence.test.ts")) continue;
      assert.ok(existsSync(join(here, rel)), `AC3: ${rel} from ${rung} is missing from summit-composed. Bring it across, or explain its change in RUNG.md (AC8) and keep a test at the same path.`);
    }
  }
});

test("AC5: RUNG.md names the four rungs and their upstream areas", () => {
  const rung = readFileSync(join(here, "RUNG.md"), "utf8");
  for (const name of rungs) assert.ok(rung.includes(name), `AC5: RUNG.md does not name ${name}.`);
  for (const path of ["apps/reader", "tests/", "packages/jsr/identity", ".github/workflows"]) {
    assert.ok(rung.includes(path), `AC5: RUNG.md's upstream mirror table does not cover ${path}.`);
  }
});

test("AC6: AGENTS.md is a short manifest with a fence and a human boundary", () => {
  const agents = readFileSync(join(here, "AGENTS.md"), "utf8");
  const words = agents.split(/\s+/).filter(Boolean).length;
  assert.ok(words < 500, `AC6: AGENTS.md has ${words} words. A Short Manifest stays under 500; move checked rules out of the prose.`);
  assert.match(agents, /^#+\s*Rung Fence/m, `AC6: AGENTS.md has no "Rung Fence" section.`);
  assert.match(agents, /^#+\s*Ask a human first/m, `AC6: AGENTS.md has no "Ask a human first" section.`);
});
~~~

Every failure names the criterion, the offender and the fix. That is a Teaching Failure, L4, the same level
you met in the variants' fences.

## Grade it

~~~bash
cd variants/summit-composed
pnpm install
pnpm test
~~~

Green means AC1 to AC6 pass. A person then reads `RUNG.md` and the history for AC7 to AC9.
